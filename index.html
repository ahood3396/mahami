مهامي Supabase

اربط التطبيق بقاعدة بياناتك السحابية

مرة واحدة فقط — يُحفظ الإعداد على هذا الجهاز، وتصبح المهام مشتركة على كل أجهزتك.

  1. أنشئ مشروعًا مجانيًا من supabase.com ثم من Project Settings → API انسخ Project URL و anon public key.
  2. من Project Settings → Authentication → Providers → Email: عطّل خيار Confirm email ليدخل المستخدمون فورًا بدون بريد تأكيد.
  3. من قائمة SQL Editor في المشروع: الصق السكربت التالي واضغط Run (ينشئ جدول المهام + سياسات الحماية + الحذف الآمن + المزامنة اللحظية).
  4. الصق الرابط والمفتاح في الحقول هنا واضغط حفظ الاتصال.
-- ================== مهامي — سكربت قاعدة البيانات ==================
create table if not exists public.tasks (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid() references auth.users(id) on delete cascade,
  title text not null,
  description text default '',
  category text default 'other',
  priority text default 'medium',
  status text default 'todo',
  due_date date,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  completed_at timestamptz
);

alter table public.tasks enable row level security;
drop policy if exists "tasks_select_own" on public.tasks;
drop policy if exists "tasks_insert_own" on public.tasks;
drop policy if exists "tasks_update_own" on public.tasks;
drop policy if exists "tasks_delete_own" on public.tasks;
create policy "tasks_select_own" on public.tasks for select using (user_id = auth.uid());
create policy "tasks_insert_own" on public.tasks for insert with check (user_id = auth.uid());
create policy "tasks_update_own" on public.tasks for update using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy "tasks_delete_own" on public.tasks for delete using (user_id = auth.uid());

-- حذف الحساب نهائيًا (يستدعي من التطبيق)
create or replace function public.mahami_delete_account()
returns void language sql security definer
set search_path = public
as $$
  delete from auth.users where id = auth.uid();
$$;

-- المزامنة اللحظية (Realtime)
do $$
begin
  if not exists (select 1 from pg_publication_tables where pubname='supabase_realtime' and tablename='tasks') then
    alter publication supabase_realtime add table public.tasks;
  end if;
end $$;
